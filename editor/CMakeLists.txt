cmake_minimum_required(VERSION 3.24)
get_filename_component(editor_real "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
if(NOT TARGET concord::concord)
    project(ConcordEditor LANGUAGES C CXX)
    get_filename_component(editor_real "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
    add_subdirectory("${editor_real}/.." "${CMAKE_CURRENT_BINARY_DIR}/engine")

file(GLOB editor_icons CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*.svg")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${editor_icons})
set(icon_header "// Generated from editor/assets; edit the SVG sources.\nnamespace Concord::Editor {\nstruct IconResource {const char* name;const char* source;};\ninline constexpr IconResource IconResources[]={\n")
foreach(icon IN LISTS editor_icons)
    get_filename_component(icon_name "${icon}" NAME_WE)
    file(READ "${icon}" icon_source)
    string(APPEND icon_header "{\"${icon_name}\",R\"SVG(${icon_source})SVG\"},\n")
endforeach()
string(APPEND icon_header "};}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/EditorIcons.gen.h" "${icon_header}")
add_executable(concord_editor WIN32
    src/Main.cpp src/Editor.cpp src/EditorChrome.cpp src/EditorHome.cpp src/EditorToolbar.cpp src/EditorPanels.cpp src/EditorViewport.cpp
    src/SceneDocument.cpp src/BuildProcess.cpp src/NativeDialogs.cpp src/Design.cpp src/EditorScripts.cpp
    src/EditorPreferences.cpp src/WorkspaceLayouts.cpp src/ConcordScriptLanguage.cpp
    src/EditorProject.cpp src/EditorSceneSettings.cpp src/ProjectDocument.cpp src/ProjectRuntime.cpp
    src/UiDesigner.cpp src/UiDesignerCanvas.cpp src/UiDesignerProperties.cpp)
target_include_directories(concord_editor PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${editor_real}/../src/3rd/hello_imgui")
target_link_libraries(concord_editor PRIVATE concord::concord ole32 uuid shell32)
if(MINGW)
    target_link_options(concord_editor PRIVATE -municode)
endif()
target_compile_definitions(concord_editor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(concord_editor PROPERTIES OUTPUT_NAME ConcordEditor)
concord_stage_runtime(concord_editor)
